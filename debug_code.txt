/**
 * JSON API for the single-page dashboard. Percent values from the services (0-100) are exposed as
 * fractions (0-1) under keys named {@code *Prob}. Money values are passed through unchanged.
 */
@RestController
@RequestMapping("/api")
public class ApiController {

    private static final long FORECAST_SEED = 42L;
    private static final int DEFAULT_HORIZON = 30;
    private static final int MAX_HORIZON = 365;
    private static final int DEFAULT_DELAY_DAYS = 15;
    private static final int MAX_DELAY_DAYS = 365;
    private static final DateTimeFormatter LABEL_FORMAT = DateTimeFormatter.ofPattern("d MMM", Locale.ENGLISH);

    /** Payment request body. Null {@code date} defaults to today, null {@code method} defaults to UPI. */
    public record PaymentRequest(BigDecimal amount, LocalDate date, PaymentMethod method) {}

    /** Invoice creation body. */
    public record InvoiceRequest(Long studentId, BigDecimal amount, LocalDate issueDate, LocalDate dueDate,
                                 String billingMonth) {}

    /** Student creation body. */
    public record StudentRequest(String name, BigDecimal monthlyFee, String notes) {}

    /** Recurring bill creation body. */
    public record BillRequest(String name, BigDecimal amount, int dayOfMonth, int graceDays) {}

    private final CashForecastService cashForecastService;
    private final RescuePlanService rescuePlanService;

    public ApiController(CashForecastService cashForecastService,
                         RescuePlanService rescuePlanService,
                         BacktestService backtestService,
                         DashboardService dashboardService,
                         DemoLock demoLock,
                         SampleDataLoader sampleDataLoader,

        this.cashForecastService = cashForecastService;

    // ---------------------------------------------------------------- dashboard

    /**
     * Dashboard payload. Keys: base, baseline, preview, openingCash, riskBannerText, rescueCards, chase, kpis, monthly.
     * {@code preview} is null unless {@code whatIfStudentId} or {@code whatIfDelayDays} is supplied.
     * {@code activeCardIds} marks matching rescue cards with {@code active: true}; it does not change any math.
     */
    @GetMapping("/dashboard")
    public Map<String, Object> dashboard(@RequestParam(defaultValue = "30") int horizon,
                                         @RequestParam(required = false) Long whatIfStudentId,
                                         @RequestParam(required = false) Integer whatIfDelayDays,
                                         @RequestParam(required = false) String activeCardIds) {
        if (horizon < 1 || horizon > MAX_HORIZON) {
            throw new IllegalArgumentException("horizon must be between 1 and " + MAX_HORIZON);
        }
        LocalDate today = LocalDate.now();
        LocalDate target = rescueTarget(today);
        // The simulator rejects a target outside its horizon, so pass it only when it falls inside.
        LocalDate horizonTarget = target != null && !target.isAfter(today.plusDays(horizon - 1L)) ? target : null;

        CashForecastResult forecast = cashForecastService.forecast(today, horizon, horizonTarget, FORECAST_SEED);
        RescuePlan plan = rescuePlanService.plan(today, horizonTarget, horizon);
        Summary summary = dashboardService.summary(today);

        Map<String, Object> base = new LinkedHashMap<>();
        List<String> labels = new ArrayList<>();
        List<Double> p10 = new ArrayList<>();
        List<Double> p50 = new ArrayList<>();
        List<Double> p90 = new ArrayList<>();
        for (DailyForecast day : forecast.days()) {
            labels.add(day.date().format(LABEL_FORMAT));
            p10.add(round2(day.worstCase().doubleValue()));
            p50.add(round2(day.mostLikely().doubleValue()));
            p90.add(round2(day.bestCase().doubleValue()));
        }
        base.put("labels", labels);
        base.put("p10", p10);
        base.put("p50", p50);
        base.put("p90", p90);
        // Forecast probability is 0-100; exposed as a fraction.
        base.put("shortfallProb", forecast.probabilityBelowZeroBeforeTarget() / 100.0);

        Map<String, Object> baseline = new LinkedHashMap<>();
        // RescuePlan.baselineShortfallPercent is 0-100; exposed as a fraction.
        baseline.put("shortfallProb", plan.baselineShortfallPercent() / 100.0);

        Object preview = null;
        if (whatIfStudentId != null || whatIfDelayDays != null) {
            int delay = clamp(whatIfDelayDays == null ? DEFAULT_DELAY_DAYS : whatIfDelayDays, 0, MAX_DELAY_DAYS);
            DelayScenarioResult scenario =
                    cashForecastService.simulateDelay(today, horizon, target, whatIfStudentId, delay);
            Map<String, Object> p = new LinkedHashMap<>();
            p.put("labels", scenario.after().labels());
            p.put("p50", roundAll(scenario.after().p50()));
            // shortfallAfter is 0-100 from CashForecastService; exposed as a fraction.
            p.put("shortfallProb", scenario.shortfallAfter() / 100.0);
            preview = p;
        }

        Set<String> active = new HashSet<>();
        if (activeCardIds != null) {
            for (String id : activeCardIds.split(",")) {
                String trimmed = id.trim();
                if (!trimmed.isEmpty()) {
                    active.add(trimmed);
                }
            }
        }
        List<Map<String, Object>> cards = new ArrayList<>();
        List<Action> actions = plan.cards();
        for (int i = 0; i < actions.size(); i++) {
            Action action = actions.get(i);
            Map<String, Object> card = new LinkedHashMap<>();
            card.put("id", action.id());
            card.put("type", action.type());
            card.put("title", action.title());
            card.put("studentId", action.studentId());
            card.put("invoiceId", action.invoiceId());
            card.put("expectedCashRecovered", action.expectedCashRecovered());
            card.put("cost", action.cost());
            card.put("reason", action.reason());
            card.put("estimate", action.estimate());
            card.put("beforeRisk", plan.baselineShortfallPercent() / 100.0);
            // riskAfterAloneByCard is aligned with cards and is 0-100; exposed as a fraction.
            card.put("afterRisk", i < plan.riskAfterAloneByCard().size()
                    ? plan.riskAfterAloneByCard().get(i) / 100.0
                    : null);
            card.put("active", active.contains(action.id()));
            cards.add(card);
        }

        List<Map<String, Object>> chase = new ArrayList<>();
        for (var suggestion : summary.chaseSuggestions()) {
            Map<String, Object> row = new LinkedHashMap<>();
            row.put("name", suggestion.studentName());
            row.put("amount", suggestion.overdueAmount());
            row.put("daysLate", suggestion.daysOverdue());
            row.put("whatsappUrl", ReminderService.whatsappUrl(suggestion.message()));
            chase.add(row);
        }

        Map<String, Object> kpis = new LinkedHashMap<>();
        kpis.put("expectedMonthlyFees", summary.expectedMonthlyFees());
        kpis.put("collectedThisMonth", summary.collectedThisMonth());
        kpis.put("pendingDues", summary.pendingDues());
        kpis.put("overdueDues", summary.overdueDues());
        kpis.put("upcomingBillsTotal", summary.upcomingBillsTotal());
        kpis.put("expectedCoverage", summary.expectedCoverage());
        kpis.put("shortfallRisk", summary.shortfallRisk());
        kpis.put("shortfallAmount", summary.shortfallAmount());

        List<Map<String, Object>> monthly = new ArrayList<>();
        for (DashboardService.MonthPoint point : summary.trend()) {
            Map<String, Object> row = new LinkedHashMap<>();
            row.put("month", point.month());
            row.put("billed", point.billed());
            row.put("collected", point.collected());
            monthly.add(row);
        }

        Map<String, Object> out = new LinkedHashMap<>();
        out.put("base", base);
        out.put("baseline", baseline);
        out.put("preview", preview);
        out.put("openingCash", forecast.openingBalance());
        out.put("riskBannerText", plan.healthy() ? plan.healthyMessage() : plan.notice());
        out.put("rescueCards", cards);
        out.put("chase", chase);
        out.put("kpis", kpis);
        out.put("monthly", monthly);
        return out;
    }

    /** Replaces data with the bad-day crisis, then returns the default dashboard. 409 if the demo is busy. */
    @PostMapping("/demo/bad-day")
    public Map<String, Object> loadBadDayDemo() {
        return runDemo(sampleDataLoader::loadBadDayDemo);
    }

    /** Restores the sample data, then returns the default dashboard. 409 if the demo is busy. */
    @PostMapping("/demo/reset")
    public Map<String, Object> resetDemo() {
        return runDemo(sampleDataLoader::resetToSampleData);
    }

    private Map<String, Object> runDemo(Runnable action) {
        if (!demoLock.tryRun(action)) {
            throw new IllegalStateException("Demo is busy");
        }
        return dashboard(DEFAULT_HORIZON, null, null, null);
    }

    // ---------------------------------------------------------------- invoices

    @GetMapping("/invoices")
    public List<Map<String, Object>> listInvoices() {
        LocalDate today = LocalDate.now();
        return invoiceService.listAll().stream().map(inv -> invoiceMap(inv, today)).toList();
    }

    @PostMapping("/invoices")
    public ResponseEntity<Map<String, Object>> createInvoice(@RequestBody InvoiceRequest request) {
        if (request == null || request.studentId() == null) {
            throw new IllegalArgumentException("studentId is required");
        }
        Student student = studentRepository.findById(request.studentId())
                .orElseThrow(() -> new IllegalArgumentException("Student not found: " + request.studentId()));
        Invoice invoice = invoiceService.createInvoice(student, request.amount(), request.issueDate(),
                request.dueDate(), request.billingMonth());
        return ResponseEntity.status(HttpStatus.CREATED).body(invoiceMap(invoice, LocalDate.now()));
    }

    /** Records a partial or full payment; the service sets PARTLY_PAID or PAID. Returns the updated invoice. */
    @PostMapping("/invoices/{id}/pay")
    public Map<String, Object> payInvoice(@PathVariable Long id, @RequestBody PaymentRequest request) {
        if (request == null) {
            throw new IllegalArgumentException("Payment body is required");
        }
        LocalDate date = request.date() == null ? LocalDate.now() : request.date();
        PaymentMethod method = request.method() == null ? PaymentMethod.UPI : request.method();
        invoiceService.recordPayment(id, request.amount(), date, method);
        return invoiceMap(invoiceService.getInvoice(id), LocalDate.now());
    }

    private static Map<String, Object> invoiceMap(Invoice invoice, LocalDate today) {
        Map<String, Object> map = new LinkedHashMap<>();
        map.put("id", invoice.getId());
        map.put("studentId", invoice.getStudent() == null ? null : invoice.getStudent().getId());
        map.put("studentName", invoice.getStudent() == null ? null : invoice.getStudent().getName());
        map.put("amount", invoice.getAmount());
        map.put("paid", invoice.amountPaid());
        map.put("dueDate", invoice.getDueDate());
        map.put("status", invoice.effectiveStatus(today).name());
        map.put("billingMonth", invoice.getBillingMonth());
        return map;
    }

    @GetMapping("/students")

    @PostMapping("/students")
    public ResponseEntity<Map<String, Object>> createStudent(@RequestBody StudentRequest request) {
        if (request == null || request.name() == null || request.name().isBlank()) {
            throw new IllegalArgumentException("Student name is required");
        }
        if (request.monthlyFee() == null || request.monthlyFee().signum() < 0) {
            throw new IllegalArgumentException("Monthly fee must be zero or positive");
        }
        Student student = studentService.create(request.name(), request.monthlyFee(), request.notes());
        return ResponseEntity.status(HttpStatus.CREATED).body(studentMap(student));
    }

    private static Map<String, Object> studentMap(Student student) {
        Map<String, Object> map = new LinkedHashMap<>();
        map.put("id", student.getId());
        map.put("name", student.getName());
        map.put("fee", student.getMonthlyFee());
        map.put("notes", student.getNotes());
        return map;
    }

    // ---------------------------------------------------------------- bills

    @GetMapping("/bills")
    public List<Map<String, Object>> listBills() {
        return recurringBillService.listAll().stream().map(ApiController::billMap).toList();
    }

    @PostMapping("/bills")
    public ResponseEntity<Map<String, Object>> createBill(@RequestBody BillRequest request) {
        if (request == null) {
            throw new IllegalArgumentException("Bill body is required");
        }
        if (request.dayOfMonth() < 1 || request.dayOfMonth() > 31) {
            throw new IllegalArgumentException("Day of month must be between 1 and 31");
        }
        if (request.graceDays() < 0) {
            throw new IllegalArgumentException("Grace days cannot be negative");
        }
        RecurringBill bill = recurringBillService.create(request.name(), request.amount(),
                request.dayOfMonth(), request.graceDays());
        return ResponseEntity.status(HttpStatus.CREATED).body(billMap(bill));
    }

    private static Map<String, Object> billMap(RecurringBill bill) {
        Map<String, Object> map = new LinkedHashMap<>();
        map.put("id", bill.getId());
        map.put("name", bill.getName());
        map.put("amount", bill.getAmount());
        map.put("dueDay", bill.getDayOfMonth());
        map.put("grace", bill.getGraceDays());
        return map;
    }

    // ---------------------------------------------------------------- backtest

    /**
     * Backtest output. Checkpoints: month, prob (predicted shortfall, 0-1), observed (actual shortfall, boolean),
     * checkpointDate. Buckets: label, checkpoints, meanPredictedProb (0-1), observedShortfallProb (0-1).
     */
    @GetMapping("/backtest")
    public Map<String, Object> backtest() {
        BacktestResult result = backtestService.runBacktest();

        List<Map<String, Object>> checkpoints = new ArrayList<>();
        for (BacktestService.Checkpoint c : result.checkpoints()) {
            Map<String, Object> row = new LinkedHashMap<>();
            row.put("checkpointDate", c.checkpointDate());
            row.put("month", c.month());
            row.put("prob", c.predictedShortfallPercent() / 100.0);
            row.put("observed", c.actualShortfall());
            checkpoints.add(row);
        }

        List<Map<String, Object>> buckets = new ArrayList<>();
        for (BacktestService.BucketRow b : result.buckets()) {
            Map<String, Object> row = new LinkedHashMap<>();
            row.put("label", b.label());
            row.put("checkpoints", b.checkpoints());
            row.put("meanPredictedProb", b.meanPredictedPercent() / 100.0);
            row.put("observedShortfallProb", b.observedShortfallPercent() / 100.0);
            buckets.add(row);
        }

        Map<String, Object> out = new LinkedHashMap<>();
        out.put("checkpoints", checkpoints);
        out.put("buckets", buckets);
        return out;
    }

    // ---------------------------------------------------------------- helpers

    /** Target date for the rescue plan: the next big bill's due date plus its grace days, or null if no bills. */
    private LocalDate rescueTarget(LocalDate today) {
        RecurringBill bill = nextBigBill(today);
        return bill == null ? null : nextDueDate(bill, today).plusDays(bill.getGraceDays());
    }

    /** The bill whose next due date is soonest (ties go to the larger amount), or null if there are no bills. */
    private RecurringBill nextBigBill(LocalDate today) {
        RecurringBill next = null;
        LocalDate nextDue = null;
        for (RecurringBill bill : recurringBillRepository.findAll()) {
            LocalDate due = nextDueDate(bill, today);
            if (next == null || due.isBefore(nextDue)
                    || (due.isEqual(nextDue) && bill.getAmount().compareTo(next.getAmount()) > 0)) {
                next = bill;
                nextDue = due;
            }
        }
        return next;
    }

    /** Next due date on or after today (grace excluded). */
    private static LocalDate nextDueDate(RecurringBill bill, LocalDate today) {
        LocalDate due = bill.dueDateFor(YearMonth.from(today));
        if (due.isBefore(today)) {
            due = bill.dueDateFor(YearMonth.from(today).plusMonths(1));
        }
        return due;
    }

    private static int clamp(int value, int min, int max) {
        return Math.max(min, Math.min(max, value));
    }

    private static List<Double> roundAll(List<Double> values) {
        List<Double> out = new ArrayList<>(values.size());
        for (Double v : values) {
            out.add(round2(v));
        }
        return out;
    }

    private static double round2(double value) {
        return BigDecimal.valueOf(value).setScale(2, RoundingMode.HALF_UP).doubleValue();
    }
}

/* Seeded PRNG (mulberry32) */
function mulberry32(a) {
  return function () {
    a |= 0; a = a + 0x6D2B79F5 | 0;
    let t = Math.imul(a ^ a >>> 15, 1 | a);
    t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t;
    return ((t ^ t >>> 14) >>> 0) / 4294967296;
  };
}

function seedState(today) {
  const y = today.getFullYear(), m = today.getMonth();
  const students = NAMES.map((_, i) => studentTemplate(i));
  const invoices = [];
  let n = 0;
  const nid = () => 'inv' + (++n);
  // 12 past PAID invoices per student (the 12 calendar months before this one)
  students.forEach((s, i) => {
    for (let k = 1; k <= 12; k++) {
      const due = new Date(y, m - k, 5);
      const late = s.behaviour === 'on-time' ? k % 3
        : s.behaviour === 'usually-late' ? 4 + ((i + k) % 8)
        : (k % 4 === 0 ? 0 : 15 + ((i * k) % 20));
      invoices.push({ id: nid(), studentId: s.id, amount: s.fee, paid: s.fee, dueDate: isoDate(due), status: 'PAID',
        payments: [{ date: isoDate(addDays(due, late)), amount: s.fee, method: METHODS[(i + k) % 3] }] });
    }
  });
  // Current-month invoices with a mix of statuses
  students.forEach((s, i) => {
    const due = new Date(y, m, 5 + (i % 3) * 5);
    let status, paid = 0, payments = [];
    if (i < 2) status = 'OVERDUE';
    else if (i < 5) { status = 'PARTLY_PAID'; paid = Math.floor(s.fee / 200) * 100; payments = [{ date: isoDate(today), amount: paid, method: METHODS[i % 3] }]; }
    else if (i < 9) status = 'SENT';
    else if (i === 9) status = 'DRAFT';
    else { status = 'PAID'; paid = s.fee; payments = [{ date: isoDate(today), amount: s.fee, method: METHODS[i % 3] }]; }
    invoices.push({ id: nid(), studentId: s.id, amount: s.fee, paid, dueDate: isoDate(due), status, payments });
  });
  return { version: 1, openingCash: 18400, students, invoices, bills: BILLS_DEFAULT.map(b => ({ ...b })) };
}

// Resolves with parsed JSON; rejects with the server's JSON "error" field when !ok.
async function api(method, path, body) {
  const opts = { method, headers: {} };
  if (body !== undefined) {
    opts.headers['Content-Type'] = 'application/json';
    opts.body = JSON.stringify(body);
  }
  const res = await fetch(path, opts);
  let data = null;
  try { data = await res.json(); } catch (e) { /* non-JSON body */ }
  if (!res.ok) throw new Error((data && data.error) || ('Request failed (' + res.status + ')'));
  return data;
}

// Occurrences of a recurring bill (day-of-month) on/after today, as day offsets.
function billOccurrences(bill, today, horizon) {
  const out = [], y = today.getFullYear(), mo = today.getMonth();
  for (let k = 0; k <= 3; k++) {
    const dim = new Date(y, mo + k + 1, 0).getDate();
    const cand = new Date(y, mo + k, Math.min(bill.dueDay, dim));
    const off = dayDiff(today, cand);
    if (off >= 0 && off <= horizon + 10) out.push({ offset: off, date: cand });
  }
  return out;
}

// Monte Carlo inputs -> scenario. Invoice inputs: outstanding, dueOffset, behaviour, chase, discount, addDelay, never.
function buildScenario(p) {
  const H = p.horizon;
  const invoices = p.invoices.map(v => {
    const b = BEHAV[v.behaviour], f = v.chase ? 0.5 : 1, add = v.addDelay || 0;
    const dMin = Math.max(0, Math.round(b.min * f) + add);
    return {
      amount: v.outstanding, dueOffset: v.dueOffset, discount: !!v.discount,
      pNon: v.never ? 1 : b.pNon * f,
      dMin, dMax: Math.max(dMin, Math.round(b.max * f) + add)
    };
  });
  const bills = [];
  let windowDays = null;
  p.bills.forEach(b => {
    const occ = billOccurrences(b, p.today, H);
    occ.forEach(o => bills.push({ name: b.name, amount: b.amount, offset: o.offset, grace: b.grace, shift: b.shift || 0 }));
    if (b.id === p.majorId && occ.length) windowDays = occ[0].offset;
  });
  return {
    horizon: H, start: p.start, dailyVar: p.dailyVar, invoices, bills,
    windowDays: Math.min(H, Math.max(1, windowDays == null ? H : windowDays))
  };
}

// One simulated path. Fixed RNG draw order so toggles/what-ifs stay comparable.
function simulateRun(scn, rng, row) {
  const H = scn.horizon, flows = new Float64Array(H + 1);
  for (const v of scn.invoices) {
    const u1 = rng(), u2 = rng(), u3 = rng();
    if (v.discount && u3 < 0.6) {            // 5% early-pay discount accepted (60%): pays in days 1â€“5
      const day = 1 + Math.floor(u3 / 0.6 * 5);
      flows[Math.min(day, H)] += v.amount * 0.95;
      continue;
    }
    if (u1 < v.pNon) continue;               // does not pay in this run
    const delay = v.dMin + Math.floor(u2 * (v.dMax - v.dMin + 1));
    const day = Math.max(1, v.dueOffset + delay);
    if (day <= H) flows[day] += v.amount;
  }
  for (const b of scn.bills) {               // paid at due day + grace uncertainty
    const u = rng();
    const day = Math.max(1, b.offset + b.shift + Math.floor(u * (b.grace + 1)));
    if (day <= H) flows[day] -= b.amount;
  }
  let bal = scn.start;
  row[0] = bal;
  for (let t = 1; t <= H; t++) {
    bal += flows[t] - scn.dailyVar * (0.5 + rng());
    row[t] = bal;
  }
}

// 1,000 runs, seed 42 -> per-day P10/P50/P90 and shortfall probability before next Rent.
function runForecast(scn, runs = 1000, seed = 42) {
  const H = scn.horizon, W = scn.windowDays, stride = H + 1;
  const buf = new Float64Array(runs * stride), rng = mulberry32(seed);
  let short = 0;
  for (let r = 0; r < runs; r++) {
    const row = buf.subarray(r * stride, (r + 1) * stride);
    simulateRun(scn, rng, row);
    let mn = Infinity;
    for (let t = 1; t <= W; t++) if (row[t] < mn) mn = row[t];
    if (mn < 0) short++;
  }
  const p10 = [], p50 = [], p90 = [], col = new Float64Array(runs);
  for (let t = 0; t <= H; t++) {
    for (let r = 0; r < runs; r++) col[r] = buf[r * stride + t];
    col.sort();
    p10.push(col[Math.floor(0.1 * (runs - 1))]);
    p50.push(col[Math.floor(0.5 * (runs - 1))]);
    p90.push(col[Math.floor(0.9 * (runs - 1))]);
  }
  return { p10, p50, p90, shortfallProb: short / runs, windowDays: W, horizon: H };
}

// Live app state -> Monte Carlo inputs. o = {horizon, whatIf:{sid,delay}|null, toggles:Set}
function liveInputs(st, today, o) {
  const wi = o.whatIf;
  const invoices = st.invoices
    .filter(i => i.status !== 'PAID' && i.amount - i.paid > 0)
    .map(i => {
      const s = st.students.find(x => x.id === i.studentId);
      if (!s) return null;
      const hit = wi && wi.sid === s.id;
      return {
        outstanding: i.amount - i.paid,
        dueOffset: dayDiff(today, parseISO(i.dueDate)),
        behaviour: s.behaviour,
        chase: o.toggles.has('chase:' + i.id),
        discount: o.toggles.has('disc:' + i.id),
        never: !!(hit && wi.delay === 'none'),
        addDelay: hit && wi.delay !== 'none' ? Number(wi.delay) : 0
      };
    }).filter(Boolean);
  const bills = st.bills.map(b => ({
    id: b.id, name: b.name, amount: b.amount, dueDay: b.dueDay, grace: b.grace,
    shift: o.toggles.has('shift:' + b.id) ? Math.min(3, b.grace) : 0
  }));
  return { today, start: st.openingCash, horizon: o.horizon, dailyVar: 250, invoices, bills, majorId: 'rent' };
}

/* ===== Backtest (synthetic 2025 checkpoints) ===== */
function runBacktest() {
  const students = NAMES.map((_, i) => studentTemplate(i));
  const rows = [];
  for (let m = 0; m < 12; m++) {
    const today = new Date(2025, m, 1);
    const invoices = students.map(s => ({ outstanding: s.fee, dueOffset: 4, behaviour: s.behaviour }));
    const start = 1000 + ((m * 2317) % 8) * 1000;
    const scn = buildScenario({ today, start, horizon: 30, dailyVar: 250, invoices, bills: BILLS_DEFAULT, majorId: 'rent' });
    const fc = runForecast(scn);
    // Observed: one synthetic realisation of the same month (separate seeded stream)
    const row = new Float64Array(scn.horizon + 1);
    simulateRun(scn, mulberry32(1000 + m), row);
    let observed = false;
    for (let t = 1; t <= scn.windowDays; t++) if (row[t] < 0) { observed = true; break; }
    const bi = BUCKETS.findIndex(b => fc.shortfallProb >= b.lo && fc.shortfallProb < b.hi);
    rows.push({ month: MONTHS_FULL[m] + ' 2025', prob: fc.shortfallProb, observed, bucket: bi });
  }
  return rows;
}

const khataPlugin = {
  id: 'khataPlugin',
  beforeDatasetsDraw(chart) {
    const { ctx, chartArea: a, scales: { y } } = chart;
    if (!a || y.min >= 0) return;
    const z = y.getPixelForValue(0);
    if (z <= a.top || z >= a.bottom) return;
    ctx.save();
    ctx.fillStyle = 'rgba(220,53,69,0.12)';            // danger zone below â‚¹0
    ctx.fillRect(a.left, z, a.right - a.left, a.bottom - z);
    ctx.restore();
  },
  afterDatasetsDraw(chart) {
    const { ctx, chartArea: a, scales: { x } } = chart;
    const marks = (chart.options.plugins.khataPlugin || {}).markers || [];
    if (!a) return;
    ctx.save();
    ctx.strokeStyle = '#6c757d'; ctx.fillStyle = '#6c757d';
    ctx.setLineDash([4, 4]); ctx.font = '11px sans-serif';
    marks.forEach(m => {
      const px = x.getPixelForValue(m.index);
      if (px < a.left || px > a.right) return;
      ctx.beginPath(); ctx.moveTo(px, a.top); ctx.lineTo(px, a.bottom); ctx.stroke();
      ctx.fillText(m.label, px + 3, a.top + 12);
    });
    ctx.restore();
  }
};

  const collected = cur.reduce((s, i) => s + Math.min(i.paid, i.amount), 0);
  const paidCount = cur.filter(i => i.status === 'PAID').length;
  const overdueInv = state.invoices.filter(i => (i.status === 'OVERDUE' || i.status === 'PARTLY_PAID') && i.amount - i.paid > 0);
  const overdueSum = overdueInv.reduce((s, i) => s + (i.amount - i.paid), 0);
  const lateStudents = new Set(overdueInv.map(i => i.studentId)).size;
  $('kpiExpected').textContent = fmt.format(expected);
  $('kpiExpectedSub').textContent = `${state.students.length} students`;
  $('kpiCollected').textContent = fmt.format(collected);
  $('kpiCollectedSub').textContent = `${paidCount} of ${cur.length} paid`;
  $('kpiOverdue').textContent = fmt.format(overdueSum);
  $('kpiOverdueSub').textContent = `${lateStudents} student${lateStudents === 1 ? '' : 's'} late`;
  $('kpiNextBill').textContent = rent ? fmt.format(rent.amount) : 'â€”';
  $('kpiNextBillSub').textContent = `Rent due on the ${ordinal(rentDay)}`;

  // Live what-if readout: shortfall without the what-if -> with it
  $('whatIfReadout').textContent = `${Math.round(baseline.shortfallProb * 100)}% â†’ ${pct}%`;
  document.querySelectorAll('[data-h]').forEach(b => b.classList.toggle('active', Number(b.dataset.h) === H));

  // Rescue Plan header badge: current risk -> risk after toggled actions
  $('riskBadge').textContent = `Risk: ${pct}% â†’ about ${pctAfter}%`;

  // Value this month: overdue balances x selected recovery assumption
  const level = RECOVERY[recoveryLevel];
  $('valueFig').textContent = fmt.format(Math.round(overdueSum * level.rate));
  $('valueNote').textContent = `${level.label} recovery: ${Math.round(level.rate * 100)}% of ${fmt.format(overdueSum)} overdue`;
  document.querySelectorAll('[data-recovery]').forEach(b => b.classList.toggle('active', b.dataset.recovery === recoveryLevel));

  renderChase(overdueInv, today);

  const labels = [];
  for (let t = 0; t <= H; t++) { const d = addDays(today, t); labels.push(`${d.getDate()} ${MON[d.getMonth()]}`); }
  const markers = [];
  state.bills.forEach(b => billOccurrences(b, today, H).forEach(o => {
    if (o.offset >= 1 && o.offset <= H) markers.push({ index: o.offset, label: b.name });
  }));
  drawChart(labels, base, preview, markers, rent.amount);

  $('rescueList').innerHTML = rescueCards().map((c, idx) => `
    <label class="kp-action">
      <span class="kp-action-num">${idx + 1}</span>
      <span class="kp-action-body"><span class="kp-action-title">${c.title}</span><span class="kp-action-gain">${c.gain}</span></span>
      <span class="kp-switch"><input class="kp-switch-input" type="checkbox" role="switch" data-tid="${c.tid}" ${rescueToggles.has(c.tid) ? 'checked' : ''}><span class="kp-switch-track"></span></span>
    </label>`).join('') || '<p class="small text-secondary mb-0">Nothing to rescue right now.</p>';
}

/* ===== Rendering: invoices ===== */
const STATUS_CLS = { PAID: 'success', PARTLY_PAID: 'warning', OVERDUE: 'danger', SENT: 'primary', DRAFT: 'secondary' };
function renderInvoices() {
  const f = $('invFilter').value;
  const rows = state.invoices
    .filter(i => !f || i.status === f)
    .sort((a, b) => b.dueDate.localeCompare(a.dueDate))
    .slice(0, 200)
    .map(i => {
      const s = state.students.find(x => x.id === i.studentId);
      const unpaid = i.status !== 'PAID';
      return `<tr>
        <td>${esc(s ? s.name : '(deleted)')}</td>
        <td class="text-end">${fmt.format(i.amount)}</td>
        <td class="text-end">${fmt.format(i.paid)}</td>
        <td>${esc(i.dueDate)}</td>
        <td><span class="badge text-bg-${STATUS_CLS[i.status] || 'secondary'}">${esc(i.status)}</span></td>
        <td class="text-nowrap">
          ${unpaid ? `<button class="btn btn-sm btn-outline-dark" data-act="pay" data-id="${esc(i.id)}">Record payment</button>
          <button class="btn btn-sm btn-outline-success" data-act="remind" data-id="${esc(i.id)}">Remind</button>` : ''}
        </td></tr>`;
    }).join('');
  $('invBody').innerHTML = rows || '<tr><td colspan="6" class="text-secondary">No invoices.</td></tr>';
}

/* ===== Rendering: backtest ===== */
async function renderBacktest() {
  if (!backtestCache) {
    try { backtestCache = await api('GET', '/api/backtest'); } catch (err) { showError(err); return; }
  }
  $('btBody').innerHTML = backtestCache.map(r => `<tr>
    <td>${esc(r.month)}</td>
    <td class="text-end">${Math.round(r.prob * 100)}%</td>
    <td>${BUCKETS[r.bucket].label}</td>
    <td>${r.observed ? 'Shortfall happened' : 'No shortfall'}</td></tr>`).join('');
  $('calBody').innerHTML = BUCKETS.map((b, i) => {
    const rows = backtestCache.filter(r => r.bucket === i);
    const hits = rows.filter(r => r.observed).length;
    return `<tr><td>${b.label}</td><td>${b.def}</td><td class="text-end">${rows.length}</td>
      <td class="text-end">${rows.length ? Math.round(hits / rows.length * 100) + '%' : 'â€”'}</td></tr>`;
  }).join('');
}

/* ===== Routing (instant, no reloads) ===== */
const VIEWS = ['dashboard', 'invoices', 'students', 'bills', 'backtest'];
function route() {
  const m = (location.hash.match(/^#\/(\w+)/) || [])[1];
  const name = VIEWS.includes(m) ? m : 'dashboard';
  VIEWS.forEach(v => $('view-' + v).classList.toggle('d-none', v !== name));
  document.querySelectorAll('[data-tab]').forEach(a => a.classList.toggle('active', a.dataset.tab === name));
  if (name === 'dashboard') renderDashboard();
  if (name === 'backtest') renderBacktest();
}

let remindDrafts = [];
function copyText(text) {
  if (navigator.clipboard && window.isSecureContext) return navigator.clipboard.writeText(text);
  const ta = document.createElement('textarea');
  ta.value = text; ta.style.position = 'fixed'; ta.style.opacity = '0';
  document.body.appendChild(ta); ta.select();
  try { document.execCommand('copy'); } catch (err) { /* ignore */ }
  ta.remove();
  return Promise.resolve();
}
function openRemind(inv) {
  const s = state.students.find(x => x.id === inv.studentId);
  const name = s ? s.name : 'there';
  const amt = fmt.format(inv.amount), due = fmtDate(parseISO(inv.dueDate));
  remindDrafts = [
    { label: 'Gentle', text: `Hi ${name}, hope you're doing well! A gentle reminder that the invoice of ${amt} was due on ${due}. Please let me know if it's already paid. Thank you! â€“ Meera` },
    { label: 'Firm', text: `Hi ${name}, the invoice of ${amt} was due on ${due} and is still pending. Please make the payment at the earliest and share the confirmation. â€“ Meera` },
    { label: 'Final', text: `Hi ${name}, this is a final reminder: the invoice of ${amt} was due on ${due} and remains unpaid. Please clear it within 2 days. â€“ Meera` }
  ];
  $('remindBody').innerHTML = remindDrafts.map((d, i) => `
    <div class="mb-3">
      <div class="d-flex justify-content-between align-items-center mb-1">
        <span class="fw-semibold">${d.label}</span>
        <span class="d-flex gap-2">
          <button class="btn btn-sm btn-outline-dark" data-copy="${i}">Copy</button>
          <a class="btn btn-sm btn-success" target="_blank" rel="noopener" href="https://wa.me/?text=${encodeURIComponent(d.text)}">Open in WhatsApp</a>
        </span>
      </div>
      <textarea class="form-control" rows="3" readonly>${esc(d.text)}</textarea>
    </div>`).join('');
  bootstrap.Modal.getOrCreateInstance($('remindModal')).show();
}
